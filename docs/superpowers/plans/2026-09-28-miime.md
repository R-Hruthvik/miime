# MIIME Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build MIIME, a lightweight Linux image viewer and annotation tool in C99 + GTK4 + Cairo that starts instantly and idles under 40 MB RAM.

**Architecture:** Non-destructive vector overlay model. Base image loaded once (immutable GdkPixbuf), annotations stored as lightweight structs in a stack, rendered via Cairo on each draw callback. On export, base + annotations are baked to a flat PNG. Event-driven redraws only — no render loop.

**Tech Stack:** C99, GTK4, Cairo, GdkPixbuf, librsvg, Pango, plain Makefile with pkg-config

**Spec:** `docs/superpowers/specs/2026-09-28-miime-design.md`

## Global Constraints

- C99 standard (`-std=c99`), compiled with `-Wall -Wextra -Werror` (warnings are errors)
- All GTK4 APIs — no GTK3 deprecated calls
- No dynamic memory without a clear free path — every `malloc`/`g_malloc` has a matching free
- No render loop — only `gtk_widget_queue_draw()` on input events
- Binary name: `miime`
- Target ~200–400 lines per `.c` file; split at ~500
- Test programs in `tests/` — simple `assert()`-based executables, no test framework dependency

```c
#include "image.h"
#include <stdlib.h>

struct MiimeImage {
    GdkPixbuf *pixbuf;
    int        width;
    int        height;
};

MiimeImage *miime_image_load(const char *path) {
    if (!path) return NULL;

    GError *err = NULL;
    GdkPixbuf *pb = gdk_pixbuf_new_from_file(path, &err);
    if (!pb) {
        if (err) {
            g_printerr("miime: failed to load '%s': %s\n", path, err->message);
            g_error_free(err);
        }
        return NULL;
    }

    MiimeImage *img = malloc(sizeof(MiimeImage));
    if (!img) {
        g_object_unref(pb);
        return NULL;
    }

    img->pixbuf = pb;
    img->width  = gdk_pixbuf_get_width(pb);
    img->height = gdk_pixbuf_get_height(pb);
    return img;
}

void miime_image_free(MiimeImage *img) {
    if (img) {
        if (img->pixbuf) {
            g_object_unref(img->pixbuf);
        }
        free(img);
    }
}

int miime_image_width(const MiimeImage *img)  { return img ? img->width  : 0; }
int miime_image_height(const MiimeImage *img) { return img ? img->height : 0; }
GdkPixbuf *miime_image_get_pixbuf(const MiimeImage *img) { return img ? img->pixbuf : NULL; }
```

```c
#include "render.h"
#include <gdk/gdk.h>

void miime_render_on_screen(cairo_t *cr,
                            MiimeImage *img,
                            double zoom,
                            double pan_x, double pan_y,
                            int widget_w, int widget_h) {
    if (!cr || !img) return;

    GdkPixbuf *pb = miime_image_get_pixbuf(img);
    if (!pb) return;

    /* Clear background to dark gray */
    cairo_set_source_rgb(cr, 0.2, 0.2, 0.2);
    cairo_paint(cr);

    /* Apply viewport transform */
    cairo_save(cr);
    cairo_translate(cr, pan_x, pan_y);
    cairo_scale(cr, zoom, zoom);

    /* Paint the image */
    GdkTexture *texture = gdk_texture_new_for_pixbuf(pb);
    /* For GTK4 + Cairo, we render via a cairo_surface from the pixbuf directly */
    int img_w = miime_image_width(img);
    int img_h = miime_image_height(img);

    cairo_surface_t *surface = cairo_image_surface_create(CAIRO_FORMAT_ARGB32, img_w, img_h);
    unsigned char *surf_data = cairo_image_surface_get_data(surface);
    int surf_stride = cairo_image_surface_get_stride(surface);

    cairo_surface_flush(surface);

    const guchar *px = gdk_pixbuf_get_pixels(pb);
    int rowstride = gdk_pixbuf_get_rowstride(pb);
    int n_channels = gdk_pixbuf_get_n_channels(pb);
    gboolean has_alpha = gdk_pixbuf_get_has_alpha(pb);

    for (int y = 0; y < img_h; y++) {
        const guchar *src_row = px + y * rowstride;
        unsigned char *dst_row = surf_data + y * surf_stride;
        for (int x = 0; x < img_w; x++) {
            guchar r = src_row[x * n_channels + 0];
            guchar g = src_row[x * n_channels + 1];
            guchar b = src_row[x * n_channels + 2];
            guchar a = has_alpha ? src_row[x * n_channels + 3] : 255;
            /* Cairo ARGB32 is premultiplied, native byte order */
            dst_row[x * 4 + 0] = (guchar)((b * a) / 255); /* B */
            dst_row[x * 4 + 1] = (guchar)((g * a) / 255); /* G */
            dst_row[x * 4 + 2] = (guchar)((r * a) / 255); /* R */
            dst_row[x * 4 + 3] = a;                        /* A */
        }
    }

    cairo_surface_mark_dirty(surface);

    cairo_set_source_surface(cr, surface, 0, 0);

    /* Use appropriate filtering */
    if (zoom > 2.0) {
        cairo_pattern_set_filter(cairo_get_source(cr), CAIRO_FILTER_NEAREST);
    } else {
        cairo_pattern_set_filter(cairo_get_source(cr), CAIRO_FILTER_BILINEAR);
    }

    cairo_paint(cr);
    cairo_surface_destroy(surface);
    g_object_unref(texture);

    cairo_restore(cr);

    (void)widget_w;
    (void)widget_h;
}
```

- [ ] **Step 4: Create `src/canvas.c`**

```c
#include "canvas.h"
#include "render.h"
#include <stdlib.h>

struct MiimeCanvas {
    GtkWidget   *drawing_area;
    MiimeImage  *image;      /* borrowed, not owned */
    double       zoom;
    double       pan_x;
    double       pan_y;
    gboolean     fit_mode;   /* TRUE = fit-to-window */
};

static void canvas_draw_cb(GtkDrawingArea *area, cairo_t *cr,
                           int width, int height, gpointer user_data) {
    MiimeCanvas *c = (MiimeCanvas *)user_data;
    (void)area;

    if (!c->image) {
        cairo_set_source_rgb(cr, 0.2, 0.2, 0.2);
        cairo_paint(cr);
        return;
    }

    double zoom = c->zoom;
    double px = c->pan_x;
    double py = c->pan_y;

    if (c->fit_mode) {
        int img_w = miime_image_width(c->image);
        int img_h = miime_image_height(c->image);
        if (img_w > 0 && img_h > 0) {
            double sx = (double)width / img_w;
            double sy = (double)height / img_h;
            zoom = (sx < sy) ? sx : sy;
            px = (width  - img_w * zoom) / 2.0;
            py = (height - img_h * zoom) / 2.0;
        }
    }

    miime_render_on_screen(cr, c->image, zoom, px, py, width, height);
}

MiimeCanvas *miime_canvas_new(void) {
    MiimeCanvas *c = malloc(sizeof(MiimeCanvas));
    if (!c) return NULL;

    c->image    = NULL;
    c->zoom     = 1.0;
    c->pan_x    = 0.0;
    c->pan_y    = 0.0;
    c->fit_mode = TRUE;

    c->drawing_area = gtk_drawing_area_new();
    gtk_widget_set_hexpand(c->drawing_area, TRUE);
    gtk_widget_set_vexpand(c->drawing_area, TRUE);

    gtk_drawing_area_set_draw_func(GTK_DRAWING_AREA(c->drawing_area),
                                   canvas_draw_cb, c, NULL);

    return c;
}

void miime_canvas_destroy(MiimeCanvas *c) {
    if (c) {
        free(c);
    }
}

void miime_canvas_set_image(MiimeCanvas *c, MiimeImage *img) {
    if (!c) return;
    c->image = img;
    c->fit_mode = TRUE;
    gtk_widget_queue_draw(c->drawing_area);
}

GtkWidget *miime_canvas_get_widget(MiimeCanvas *c) {
    return c ? c->drawing_area : NULL;
}
```

```c
static gboolean canvas_scroll_cb(GtkEventControllerScroll *ctrl,
                                 double dx, double dy,
                                 gpointer user_data) {
    MiimeCanvas *c = (MiimeCanvas *)user_data;
    (void)ctrl;
    (void)dx;

    if (!c->image) return FALSE;

    /* Exit fit mode on first manual zoom */
    if (c->fit_mode) {
        /* Compute current fit zoom and pan to use as starting point */
        int widget_w = gtk_widget_get_width(c->drawing_area);
        int widget_h = gtk_widget_get_height(c->drawing_area);
        int img_w = miime_image_width(c->image);
        int img_h = miime_image_height(c->image);
        double sx = (double)widget_w / img_w;
        double sy = (double)widget_h / img_h;
        c->zoom = (sx < sy) ? sx : sy;
        c->pan_x = (widget_w - img_w * c->zoom) / 2.0;
        c->pan_y = (widget_h - img_h * c->zoom) / 2.0;
        c->fit_mode = FALSE;
    }

    /* Get cursor position relative to widget */
    /* For scroll events, we need to track cursor position separately */
    double mouse_x = c->last_mouse_x;
    double mouse_y = c->last_mouse_y;

    /* Zoom factor: scroll up = zoom in, scroll down = zoom out */
    double factor = (dy < 0) ? 1.1 : 1.0 / 1.1;
    double new_zoom = c->zoom * factor;

    /* Clamp zoom to reasonable range */
    if (new_zoom < 0.01) new_zoom = 0.01;
    if (new_zoom > 100.0) new_zoom = 100.0;

    /* Anchor zoom to cursor position */
    c->pan_x = mouse_x - (mouse_x - c->pan_x) * (new_zoom / c->zoom);
    c->pan_y = mouse_y - (mouse_y - c->pan_y) * (new_zoom / c->zoom);
    c->zoom = new_zoom;

    gtk_widget_queue_draw(c->drawing_area);
    return TRUE;
}
```

```c
#include <string.h>
#include <dirent.h>

struct MiimeDirList {
    char **paths;       /* array of full paths */
    int    count;
    int    current;     /* index of current image */
};

static int str_compare(const void *a, const void *b) {
    return strcmp(*(const char **)a, *(const char **)b);
}

static gboolean is_image_extension(const char *name) {
    const char *ext = strrchr(name, '.');
    if (!ext) return FALSE;
    ext++; /* skip the dot */
    const char *supported[] = {"png","jpg","jpeg","webp","bmp","gif","tiff","tif","svg",NULL};
    for (int i = 0; supported[i]; i++) {
        if (g_ascii_strcasecmp(ext, supported[i]) == 0) return TRUE;
    }
    return FALSE;
}

MiimeDirList *miime_dir_list_new(const char *image_path) {
    if (!image_path) return NULL;

    char *path_copy = g_strdup(image_path);
    char *dir = g_path_get_dirname(path_copy);
    char *basename = g_path_get_basename(path_copy);

    DIR *d = opendir(dir);
    if (!d) {
        g_free(path_copy); g_free(dir); g_free(basename);
        return NULL;
    }

    MiimeDirList *dl = malloc(sizeof(MiimeDirList));
    dl->paths = NULL;
    dl->count = 0;
    dl->current = 0;

    struct dirent *entry;
    int capacity = 64;
    dl->paths = malloc(sizeof(char *) * capacity);

    while ((entry = readdir(d)) != NULL) {
        if (entry->d_name[0] == '.') continue;
        if (!is_image_extension(entry->d_name)) continue;

        if (dl->count >= capacity) {
            capacity *= 2;
            dl->paths = realloc(dl->paths, sizeof(char *) * capacity);
        }
        dl->paths[dl->count] = g_build_filename(dir, entry->d_name, NULL);
        dl->count++;
    }
    closedir(d);

    /* Sort alphabetically */
    qsort(dl->paths, dl->count, sizeof(char *), str_compare);

    /* Find current image index */
    char *full_path = g_build_filename(dir, basename, NULL);
    for (int i = 0; i < dl->count; i++) {
        if (strcmp(dl->paths[i], full_path) == 0) {
            dl->current = i;
            break;
        }
    }
    g_free(full_path);
    g_free(path_copy);
    g_free(dir);
    g_free(basename);

    return dl;
}

void miime_dir_list_free(MiimeDirList *dl) {
    if (!dl) return;
    for (int i = 0; i < dl->count; i++) g_free(dl->paths[i]);
    free(dl->paths);
    free(dl);
}

const char *miime_dir_list_current(MiimeDirList *dl) {
    if (!dl || dl->count == 0) return NULL;
    return dl->paths[dl->current];
}

const char *miime_dir_list_next(MiimeDirList *dl) {
    if (!dl || dl->count == 0) return NULL;
    dl->current = (dl->current + 1) % dl->count;
    return dl->paths[dl->current];
}

const char *miime_dir_list_prev(MiimeDirList *dl) {
    if (!dl || dl->count == 0) return NULL;
    dl->current = (dl->current - 1 + dl->count) % dl->count;
    return dl->paths[dl->current];
}
```

```c
#include "annotate.h"
#include <stdlib.h>
#include <string.h>

struct MiimeAnnotationStack {
    MiimeAnnotation *items;
    int capacity;
    int total;    /* total items in buffer (including undone) */
    int active;   /* number of active (non-undone) items, active <= total */
};

MiimeAnnotationStack *miime_annot_stack_new(void) {
    MiimeAnnotationStack *s = malloc(sizeof(MiimeAnnotationStack));
    if (!s) return NULL;
    s->capacity = 64;
    s->items = malloc(sizeof(MiimeAnnotation) * s->capacity);
    s->total = 0;
    s->active = 0;
    return s;
}

void miime_annot_stack_free(MiimeAnnotationStack *s) {
    if (!s) return;
    /* Free any blur_data in annotations */
    for (int i = 0; i < s->total; i++) {
        if (s->items[i].blur_data) {
            free(s->items[i].blur_data);
        }
    }
    free(s->items);
    free(s);
}

void miime_annot_stack_push(MiimeAnnotationStack *s, MiimeAnnotation annot) {
    if (!s) return;

    /* Discard any undone items (clear redo stack) */
    for (int i = s->active; i < s->total; i++) {
        if (s->items[i].blur_data) {
            free(s->items[i].blur_data);
            s->items[i].blur_data = NULL;
        }
    }
    s->total = s->active;

    /* Grow if needed */
    if (s->total >= s->capacity) {
        s->capacity *= 2;
        s->items = realloc(s->items, sizeof(MiimeAnnotation) * s->capacity);
    }

    s->items[s->total] = annot;
    s->total++;
    s->active = s->total;
}

gboolean miime_annot_stack_undo(MiimeAnnotationStack *s) {
    if (!s || s->active == 0) return FALSE;
    s->active--;
    return TRUE;
}

gboolean miime_annot_stack_redo(MiimeAnnotationStack *s) {
```

```c
static void render_one_annotation(cairo_t *cr, const MiimeAnnotation *a) {
    if (!a) return;

    cairo_set_source_rgba(cr, a->color_r, a->color_g, a->color_b, a->color_a);
    cairo_set_line_width(cr, a->stroke_width);
    cairo_set_line_cap(cr, CAIRO_LINE_CAP_ROUND);
    cairo_set_line_join(cr, CAIRO_LINE_JOIN_ROUND);

    switch (a->type) {
        case MIIME_ANNOT_PEN:
            if (a->point_count < 4) break;
            cairo_move_to(cr, a->points[0], a->points[1]);
            for (int i = 2; i < a->point_count; i += 2) {
                cairo_line_to(cr, a->points[i], a->points[i + 1]);
            }
            cairo_stroke(cr);
            break;

        case MIIME_ANNOT_LINE:
            if (a->point_count < 4) break;
            cairo_move_to(cr, a->points[0], a->points[1]);
            cairo_line_to(cr, a->points[2], a->points[3]);
            cairo_stroke(cr);
            break;

        case MIIME_ANNOT_RECT: {
            if (a->point_count < 4) break;
            double x = (a->points[0] < a->points[2]) ? a->points[0] : a->points[2];
            double y = (a->points[1] < a->points[3]) ? a->points[1] : a->points[3];
            double w = (a->points[0] < a->points[2]) ? a->points[2] - a->points[0]
                                                       : a->points[0] - a->points[2];
            double h = (a->points[1] < a->points[3]) ? a->points[3] - a->points[1]
                                                       : a->points[1] - a->points[3];
            cairo_rectangle(cr, x, y, w, h);
            cairo_stroke(cr);
            break;
        }

        case MIIME_ANNOT_ELLIPSE: {
            if (a->point_count < 4) break;
            double cx = (a->points[0] + a->points[2]) / 2.0;
            double cy = (a->points[1] + a->points[3]) / 2.0;
            double rx = (a->points[2] - a->points[0]) / 2.0;
            double ry = (a->points[3] - a->points[1]) / 2.0;
            if (rx < 0) rx = -rx;
            if (ry < 0) ry = -ry;
            if (rx < 1 || ry < 1) break;
            cairo_save(cr);
            cairo_translate(cr, cx, cy);
            cairo_scale(cr, rx, ry);
            cairo_arc(cr, 0, 0, 1.0, 0, 2 * G_PI);
            cairo_restore(cr);
            cairo_stroke(cr);
            break;
        }

        case MIIME_ANNOT_ARROW: {
            if (a->point_count < 4) break;
            double x1 = a->points[0], y1 = a->points[1];
            double x2 = a->points[2], y2 = a->points[3];
            /* Line */
            cairo_move_to(cr, x1, y1);
            cairo_line_to(cr, x2, y2);
            cairo_stroke(cr);
            /* Arrowhead */
            double angle = atan2(y2 - y1, x2 - x1);
            double head_len = a->stroke_width * 4;
            double head_angle = 0.4; /* ~23 degrees */
            double ax = x2 - head_len * cos(angle - head_angle);
            double ay = y2 - head_len * sin(angle - head_angle);
            double bx = x2 - head_len * cos(angle + head_angle);
            double by = y2 - head_len * sin(angle + head_angle);
            cairo_move_to(cr, x2, y2);
            cairo_line_to(cr, ax, ay);
            cairo_move_to(cr, x2, y2);
            cairo_line_to(cr, bx, by);
            cairo_stroke(cr);
            break;
        }

        case MIIME_ANNOT_TEXT:
        case MIIME_ANNOT_BLUR:
        case MIIME_ANNOT_CROP:
            /* Implemented in later tasks */
            break;
    }
}

void miime_render_annotations(cairo_t *cr,
                              const MiimeAnnotationStack *stack,
                              const MiimeAnnotation *preview) {
    if (!cr) return;

    if (stack) {
        int count = miime_annot_stack_count(stack);
        for (int i = 0; i < count; i++) {
            render_one_annotation(cr, miime_annot_stack_get(stack, i));
        }
    }

    if (preview) {
        render_one_annotation(cr, preview);
    }
}
```

```c
/* In tools.c — new function: */
void miime_tool_compute_blur(MiimeToolState *ts, MiimeImage *img) {
    if (!ts || !ts->active || ts->type != MIIME_TOOL_BLUR) return;

    MiimeAnnotation *a = &ts->current;
    double x = (a->points[0] < a->points[2]) ? a->points[0] : a->points[2];
    double y = (a->points[1] < a->points[3]) ? a->points[1] : a->points[3];
    double w = fabs(a->points[2] - a->points[0]);
    double h = fabs(a->points[3] - a->points[1]);

    int bx = (int)x, by = (int)y, bw = (int)w, bh = (int)h;
    if (bw < 1 || bh < 1) return;

    /* Clamp to image bounds */
    int img_w = miime_image_width(img);
    int img_h = miime_image_height(img);
    if (bx < 0) bx = 0;
    if (by < 0) by = 0;
    if (bx + bw > img_w) bw = img_w - bx;
    if (by + bh > img_h) bh = img_h - by;
    if (bw < 1 || bh < 1) return;

    GdkPixbuf *pb = miime_image_get_pixbuf(img);
    const guchar *pixels = gdk_pixbuf_get_pixels(pb);
    int rowstride = gdk_pixbuf_get_rowstride(pb);
    int n_ch = gdk_pixbuf_get_n_channels(pb);

    int factor = 8;
    a->blur_width = bw;
    a->blur_height = bh;
    a->blur_stride = bw * 4;
    a->blur_data = malloc(bw * bh * 4);

    /* Pixelate: for each factor×factor block, average colors */
    for (int py = 0; py < bh; py++) {
        for (int px = 0; px < bw; px++) {
            int block_x = (px / factor) * factor;
            int block_y = (py / factor) * factor;
            /* Average the block */
            int r = 0, g = 0, b = 0, count = 0;
            for (int dy = 0; dy < factor && (block_y + dy) < bh; dy++) {
                for (int dx = 0; dx < factor && (block_x + dx) < bw; dx++) {
                    int sx = bx + block_x + dx;
                    int sy = by + block_y + dy;
                    if (sx < img_w && sy < img_h) {
                        const guchar *p = pixels + sy * rowstride + sx * n_ch;
                        r += p[0]; g += p[1]; b += p[2];
                        count++;
                    }
                }
            }
            if (count > 0) {
                r /= count; g /= count; b /= count;
            }
            int idx = (py * bw + px) * 4;
            a->blur_data[idx + 0] = (unsigned char)b; /* B - Cairo ARGB32 */
            a->blur_data[idx + 1] = (unsigned char)g; /* G */
            a->blur_data[idx + 2] = (unsigned char)r; /* R */
            a->blur_data[idx + 3] = 255;               /* A */
        }
    }

    /* Store bounding box position */
    a->points[0] = bx;
    a->points[1] = by;
    a->points[2] = bx + bw;
    a->points[3] = by + bh;
}
```

```c
cairo_surface_t *miime_render_bake(MiimeImage *img,
                                   const MiimeAnnotationStack *stack) {
    if (!img) return NULL;

    int w = miime_image_width(img);
    int h = miime_image_height(img);

    /* Check for crop annotation — use last one if present */
    int crop_x = 0, crop_y = 0, crop_w = w, crop_h = h;
    if (stack) {
        int count = miime_annot_stack_count(stack);
        for (int i = count - 1; i >= 0; i--) {
            const MiimeAnnotation *a = miime_annot_stack_get(stack, i);
            if (a->type == MIIME_ANNOT_CROP && a->point_count >= 4) {
                crop_x = (int)((a->points[0] < a->points[2]) ? a->points[0] : a->points[2]);
                crop_y = (int)((a->points[1] < a->points[3]) ? a->points[1] : a->points[3]);
                crop_w = (int)fabs(a->points[2] - a->points[0]);
                crop_h = (int)fabs(a->points[3] - a->points[1]);
                break;
            }
        }
    }

    cairo_surface_t *surface = cairo_image_surface_create(CAIRO_FORMAT_ARGB32, crop_w, crop_h);
    cairo_t *cr = cairo_create(surface);

    /* Offset to crop origin */
    cairo_translate(cr, -crop_x, -crop_y);

    /* Paint base image at 1:1 */
    /* (reuse the pixel conversion code from miime_render_on_screen) */
    miime_render_on_screen(cr, img, stack, NULL, 1.0, 0, 0, w, h);

    cairo_destroy(cr);
    return surface;
}

gboolean miime_render_save_png(MiimeImage *img,
                               const MiimeAnnotationStack *stack,
                               const char *path) {
    cairo_surface_t *surface = miime_render_bake(img, stack);
    if (!surface) return FALSE;
    cairo_status_t status = cairo_surface_write_to_png(surface, path);
    cairo_surface_destroy(surface);
    return (status == CAIRO_STATUS_SUCCESS);
}
```

```c
#include "clipboard.h"
#include "render.h"
#include <cairo.h>

void miime_clipboard_copy(MiimeImage *img,
                          const MiimeAnnotationStack *stack,
                          GdkDisplay *display) {
    if (!img || !display) return;

    cairo_surface_t *surface = miime_render_bake(img, stack);
    if (!surface) return;

    int w = cairo_image_surface_get_width(surface);
    int h = cairo_image_surface_get_height(surface);

    /* Convert Cairo surface to GdkPixbuf for clipboard */
    unsigned char *data = cairo_image_surface_get_data(surface);
    int stride = cairo_image_surface_get_stride(surface);

    GdkPixbuf *pb = gdk_pixbuf_new(GDK_COLORSPACE_RGB, TRUE, 8, w, h);
    guchar *pb_pixels = gdk_pixbuf_get_pixels(pb);
    int pb_stride = gdk_pixbuf_get_rowstride(pb);

    for (int y = 0; y < h; y++) {
        unsigned char *src = data + y * stride;
        guchar *dst = pb_pixels + y * pb_stride;
        for (int x = 0; x < w; x++) {
            /* Cairo ARGB32 (premultiplied) → GdkPixbuf RGBA */
            guchar b = src[x * 4 + 0];
            guchar g = src[x * 4 + 1];
            guchar r = src[x * 4 + 2];
            guchar a = src[x * 4 + 3];
            if (a > 0) {
                dst[x * 4 + 0] = (guchar)((r * 255) / a);
                dst[x * 4 + 1] = (guchar)((g * 255) / a);
                dst[x * 4 + 2] = (guchar)((b * 255) / a);
            } else {
                dst[x * 4 + 0] = dst[x * 4 + 1] = dst[x * 4 + 2] = 0;
            }
            dst[x * 4 + 3] = a;
        }
    }

    GdkClipboard *clipboard = gdk_display_get_clipboard(display);
    GdkTexture *texture = gdk_texture_new_for_pixbuf(pb);
    gdk_clipboard_set_texture(clipboard, texture);

    g_object_unref(texture);
    g_object_unref(pb);
    cairo_surface_destroy(surface);
}
```
